auth.signInWithEmailAndPassword(email, pass) .then(async (cred) => { const uid = cred.user.uid; const snap = await db.ref("usuarios/" + uid).once("value"); const data = snap.val(); if (!data) { setStatus("SIN PERFIL DE USUARIO", "red"); auth.signOut(); reset(); return; } // Normalizamos el rol a mayúsculas para evitar errores de coincidencia const rol = data.rol ? data.rol.toUpperCase() : "JUGADOR"; const usuario = { id: uid, email, callsign: data.callsign || "SIN_ALIAS", rol: rol }; localStorage.setItem("usuario", JSON.stringify(usuario)); setStatus("ACCESO CONCEDIDO", "lime"); setTimeout(() => { // NUEVA LÓGICA DE REDIRECCIÓN SOLICITADA if (rol === "ADMIN" || rol === "EDITOR") { // Si es ADMIN o EDITOR -> hub2.html location.href = "hub2.html"; } else { // Si es JUGADOR (o cualquier otro) -> hub.html location.href = "hub.html"; } }, 600); }) .catch(err => { setStatus("ERROR DE ACCESO", "red"); console.error(err); reset(); });
