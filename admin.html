function seleccionarMision(id) { miMisionEdicionId = id; limpiarMapa(); document.getElementById('btn-save').classList.add('hidden'); document.getElementById('btn-update').classList.remove('hidden'); db.ref(`operaciones/${id}`).once('value', snap => { const m = snap.val(); document.getElementById('op-name').value = m.nombre; // Creamos un objeto para calcular los límites del zoom const bounds = new google.maps.LatLngBounds(); m.zonas.forEach(z => { let shape; if(z.type === 'polygon') { shape = new google.maps.Polygon({ paths: z.path, map: map, editable: true }); // Expandir límites con cada punto del polígono z.path.forEach(punto => bounds.extend(punto)); } else { shape = new google.maps.Marker({ position: {lat: z.lat, lng: z.lng}, map: map, draggable: true }); // Expandir límites con la posición del marcador bounds.extend({lat: z.lat, lng: z.lng}); } shape.role = z.role; shape.type = z.type; aplicarEstilo(shape, z.role); allShapes.push(shape); }); // Ajustar el mapa a los elementos cargados if (!bounds.isEmpty()) { map.fitBounds(bounds); // Opcional: Evitar que el zoom sea demasiado cercano si hay pocos elementos const listener = google.maps.event.addListener(map, 'idle', () => { if (map.getZoom() > 18) map.setZoom(18); google.maps.event.removeListener(listener); }); } escucharInscritos(id); }); }
